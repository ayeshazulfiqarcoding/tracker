{
  "version": "2.0.0",
  "tasks": [
    {
      "type": "shell",
      "label": "Build C++20 with Clang",
      "command": "/usr/bin/clang++",
      "args": [
        "-std=c++20",
        "-Wall",
        "-Wextra",
        "-g",
        "${file}",
        "-o",
        "${fileDirname}/${fileBasenameNoExtension}"
      ],
      "options": {
        "cwd": "${fileDirname}"
      },
      "problemMatcher": ["$gcc"],
      "group": {
        "kind": "build",
        "isDefault": true
      }
    }
  ]
}

Starter C++20 Console App Code (`main.cpp`)

#include <iostream>
#include <vector>
#include <string>
#include <fstream>
#include <algorithm>

// Cyber & Full-Stack Command Center - Starter CLI Module
class TaskManager {
private:
    std::vector<std::string> tasks;

public:
    void addTask(const std::string& task) {
        tasks.push_back(task);
        std::cout << "[+] Task Added: " << task << "\n";
    }

    void displayTasks() const {
        std::cout << "\n=== CURRENT SPRINT TASKS ===\n";
        for (size_t i = 0; i < tasks.size(); ++i) {
            std::cout << " [" << (i + 1) << "] " << tasks[i] << "\n";
        }
    }
};

int main() {
    std::cout << "=========================================\n";
    std::cout << "  Cyber Command Center CLI (macOS C++20)  \n";
    std::cout << "=========================================\n";

    TaskManager manager;
    manager.addTask("Configure macOS Clang Toolchain");
    manager.addTask("Complete LearnCpp Pointers Chapter");
    manager.addTask("Audit TryHackMe Linux Fundamentals");
    
    manager.displayTasks();

    return 0;
}
