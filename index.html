      3. Day 1 Starter: Log Analyzer (Standard Headers Only)

      Uses strictly <iostream>, <vector>, <string>, and <algorithm>.

      #include <iostream>
#include <vector>
#include <string>
#include <algorithm>

struct AuditEvent {
    std::string host;
    std::string status;
    int port;
};

int main() {
    std::vector<AuditEvent> events = {
        {"192.168.1.10", "OPEN", 22},
        {"192.168.1.14", "FILTERED", 443},
        {"192.168.1.22", "OPEN", 8080}
    };

    int openCount = 0;
    for (const auto& ev : events) {
        if (ev.status == "OPEN") {
            ++openCount;
            std::cout << "[ALERT] Host " << ev.host
                      << " open on port " << ev.port << '\n';
        }
    }
    std::cout << "Total open ports: " << openCount << '\n';
    return 0;
}
