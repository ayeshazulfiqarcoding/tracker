C++20

Setup checklist

Compile and run

clang++ -std=c++20 -Wall -Wextra -g main.cpp -o main && ./main

With sanitizers and leak check

clang++ -std=c++20 -g -fsanitize=address,undefined main.cpp -o main && ./main
leaks --atExit -- ./main

Starter console app main.cpp

#include <iostream>
#include <vector>
#include <string>
#include <fstream>
#include <algorithm>

// Cyber Command Center: starter CLI module
class TaskManager {
private:
    std::vector<std::string> tasks;

public:
    void addTask(const std::string& task) {
        tasks.push_back(task);
        std::cout << "[+] Task Added: " << task << "\n";
    }

    void displayTasks() const {
        std::cout << "\n=== CURRENT SPRINT TASKS ===\n";
        for (size_t i = 0; i < tasks.size(); ++i) {
            std::cout << " [" << (i + 1) << "] " << tasks[i] << "\n";
        }
    }
};

int main() {
    std::cout << "=========================================\n";
    std::cout << "  Cyber Command Center CLI (macOS C++20)  \n";
    std::cout << "=========================================\n";

    TaskManager manager;
    manager.addTask("Configure macOS Clang Toolchain");
    manager.addTask("Complete LearnCpp Pointers Chapter");
    manager.addTask("Audit TryHackMe Linux Fundamentals");

    manager.displayTasks();
    return 0;
}

Day 3 drill: references, pointers, ownership

#include <iostream>
#include <memory>
#include <vector>

void bump(int& value) { ++value; }

int main() {
    int n = 41;
    bump(n);                                  // reference: modifies n
    int* p = &n;                              // pointer: holds an address
    auto owned = std::make_unique<int>(*p);   // unique_ptr: frees itself (RAII)
    std::vector<int> v{1, 2, 3};
    v.push_back(*owned);
    std::cout << n << ' ' << v.back() << '\n';   // prints: 42 42
}
